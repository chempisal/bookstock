រក្សាទុកទិន្នន័យសៀវភៅ សិស្ស និងការខ្ចី-សងដោយស្វ័យប្រវត្ត ឬដោយដៃ
ស្ថានភាព៖ មិនទាន់ភ្ជាប់
🛠️ របៀបតម្លើង និងរៀបចំ Google Sheets (ចុចទីនេះ)
  1. បង្កើត Google Sheet ថ្មីមួយនៅក្នុងគណនី Google Drive របស់អ្នក។
  2. បើក Google Sheet នោះ រួចចូលទៅកាន់ Extensions (ផ្នែកបន្ថែម) -> Apps Script។
  3. លុបកូដចាស់ៗចោល រួចចម្លងកូដ (Apps Script) ខាងក្រោមនេះទៅដាក់៖
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Save books
  var sheetBooks = ss.getSheetByName("Books") || ss.insertSheet("Books");
  sheetBooks.clear();
  sheetBooks.appendRow(["ID", "Title", "Subject", "Grade", "Quantity", "Borrowed"]);
  data.books.forEach(function(b) {
    sheetBooks.appendRow([b.id, b.title, b.subject, b.grade, b.quantity, b.borrowed]);
  });
  
  // Save students
  var sheetStudents = ss.getSheetByName("Students") || ss.insertSheet("Students");
  sheetStudents.clear();
  sheetStudents.appendRow(["ID", "Name", "Gender", "Grade", "Class"]);
  data.students.forEach(function(s) {
    sheetStudents.appendRow([s.id, s.name, s.gender, s.grade, s.class]);
  });
  
  // Save borrowings
  var sheetBorrowings = ss.getSheetByName("Borrowings") || ss.insertSheet("Borrowings");
  sheetBorrowings.clear();
  sheetBorrowings.appendRow(["ID", "Student Name", "Student ID", "Grade", "Class", "Borrow Date", "Due Date", "Status", "Books (JSON)"]);
  data.borrowings.forEach(function(br) {
    sheetBorrowings.appendRow([
      br.id, 
      br.studentName, 
      br.studentId || "", 
      br.grade, 
      br.class, 
      br.borrowDate, 
      br.dueDate, 
      br.status, 
      JSON.stringify(br.books)
    ]);
  });
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}

  2. ជ្រើសរើសប្រភេទ Web app (កម្មវិធីបណ្តាញ) ដោយកំណត់៖
    • Execute as (ដំណើរការជា)៖ Me (គណនីរបស់អ្នក)
    • Who has access (អ្នកណាខ្លះអាចចូលប្រើ)៖ Anyone (អ្នកណាក៏ដោយ)
